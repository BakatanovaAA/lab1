# Работа номер 1
## Код
``` c
#include <locale.h>
#include <stdio.h>

void name()
{
    puts("************************************************");
    puts("*                                              *");
    puts("*   Тема: Разработка консольного приложения    *");
    puts("*                                              *");
    puts("*         Выполнила Бакатанова А.В.            *");
    puts("*                                              *");
    puts("************************************************");
}

void date()
{
    puts(" _   _     _         _   _ ");
    puts("| | |_|   | |   |   | | |_|");
    puts("|_|  _|.  |_|   |.  |_| |_|");
}

int main()
{
    setlocale(LC_ALL, "RUS");
    name();
    date();
    getchar();
    return 0;
}
```
